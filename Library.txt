        let selectedLang = 'all';
        let selectedType = 'all';

        // Direct search routing logic 
        function executeExternalSearch() {
            const queryInput = document.getElementById('searchQuery');
            const engineSelect = document.getElementById('searchEngine');
            const resultArea = document.getElementById('linkOutputArea');

            const queryText = queryInput.value.trim();
            const engine = engineSelect.value;

            if (!queryText) {
                alert("Please enter a search term first.");
                return;
            }

            let targetUrl = "";
            let engineName = "";

            switch(engine) {
                case "google":
                    targetUrl = `https://scholar.google.com/scholar?q=${encodeURIComponent(queryText)}`;
                    engineName = "Google Scholar";
                    break;
                case "semantic":
                    targetUrl = `https://www.semanticscholar.org/search?q=${encodeURIComponent(queryText)}`;
                    engineName = "Semantic Scholar";
                    break;
                case "archive_org":
                    targetUrl = `https://archive.org/details/books?and[]=${encodeURIComponent(queryText)}`;
                    engineName = "Internet Archive Books";
                    break;
                case "core":
                    targetUrl = `https://core.ac.uk/search?q=${encodeURIComponent(queryText)}`;
                    engineName = "CORE Academic";
                    break;
                case "base_search":
                    targetUrl = `https://www.base-search.net/Search/Results?lookfor=${encodeURIComponent(queryText)}`;
                    engineName = "BASE Search";
                    break;
                case "pubmed":
                    targetUrl = `https://pubmed.ncbi.nlm.nih.gov/?term=${encodeURIComponent(queryText)}`;
                    engineName = "PubMed";
                    break;
                case "researchgate":
                    targetUrl = `https://www.researchgate.net/search/publication?q=${encodeURIComponent(queryText)}`;
                    engineName = "ResearchGate";
                    break;
            }
            
            window.open(targetUrl, '_blank');

            resultArea.innerHTML = `
                <a href="${targetUrl}" target="_blank" class="persistent-link-btn">
                    □ Return to ${engineName} search results for: "${queryText}"
                </a>
            `;
        }

        document.getElementById('searchQuery').addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                executeExternalSearch();
            }
        });

        function renderGrid() {
            const grid = document.getElementById('libraryGrid');
            const counter = document.getElementById('counterDisplay');
            grid.innerHTML = '';

            const filtered = libraries.filter(item => {
                const langMatch = selectedLang === 'all' || item.lang === selectedLang;
                const typeMatch = selectedType === 'all' || item.type === selectedType;
                return langMatch && typeMatch;
            });

            counter.innerText = `Showing ${filtered.length} of ${libraries.length} verified repositories available.`;

            if (filtered.length === 0) {
                grid.innerHTML = `<p style="grid-column: 1/-1; text-align: center; color: var(--text-muted); padding: 4rem; font-size: 1.1rem;">No digital library frameworks match your active filters.</p>`;
                return;
            }

            filtered.forEach(item => {
                const card = document.createElement('div');
                card.className = `lib-card card-${item.type}`;
                card.innerHTML = `
                    <div>
                        <div class="lib-meta">
                            <div class="badges-wrapper">
                                ${item.type === 'islamic' ? '<span class="type-badge">□ Islamic</span>' : ''}
                                <span class="lang-badge lang-${item.lang}">${item.lang}</span>
                            </div>
                        </div>
                        <div class="lib-name">${item.name}</div>
                        <p class="lib-desc">${item.description}</p>
                        <div class="card-meta-tags">
                            ${item.tags.map(t => `<span class="meta-tag">${t}</span>`).join('')}
                        </div>
                    </div>
                    <a href="${item.url}" target="_blank" class="visit-btn">Open Repository</a>
                `;
                grid.appendChild(card);
            });
        }

        function filterLanguage(lang) {
            selectedLang = lang;
            document.querySelectorAll('[id^="lang-"]').forEach(btn => btn.classList.remove('active'));
            document.getElementById(`lang-${lang}`).classList.add('active');
            renderGrid();
        }

        function filterType(type) {
            selectedType = type;
            document.querySelectorAll('[id^="type-"]').forEach(btn => btn.classList.remove('active'));
            document.getElementById(`type-${type}`).classList.add('active');
            renderGrid();
        }

        renderGrid();
